"""Direct, read-only Search probes for saved answers; oracle hints are diagnostic only."""

import argparse
import asyncio
import hashlib
import json
import re
import sys
import time
from collections import Counter
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

from azure.identity.aio import DefaultAzureCredential
from azure.search.documents.aio import SearchClient
from azure.search.documents.models import VectorizableTextQuery

from crv_agent.adapters.search import SELECT, query_filter

STATUSES = {"partial", "fail", "needs_review"}
REPORT_CODE = re.compile(r"\bCR(?:/[A-Z0-9-]+){2,}/\d{2}\b", re.IGNORECASE)


def probe_queries(row: dict[str, Any]) -> list[dict[str, Any]]:
    reference = row["expected_answer"]
    codes = list(dict.fromkeys(REPORT_CODE.findall(reference)))
    hint = " OR ".join('"' + code.replace('"', '\\"') + '"' for code in codes)
    return [
        {"arm": "question_lexical", "text": row["question"], "oracle": False},
        {"arm": "question_hybrid", "text": row["question"], "oracle": False},
        {"arm": "reference_lookup", "text": hint or reference, "oracle": True},
    ]


async def query(
    client: SearchClient, row: dict[str, Any], probe: dict[str, Any], release: str, top: int,
) -> dict[str, Any]:
    started = time.perf_counter()
    result = {"id": row["id"], **probe, "release": release}
    try:
        hybrid = probe["arm"] == "question_hybrid"
        async with asyncio.timeout(90):
            rows = await client.search(
                search_text=probe["text"], filter=query_filter(release, []),
                select=SELECT, top=top, include_total_count=True,
                query_type="semantic" if hybrid else "simple",
                semantic_configuration_name="crv-evidence-semantic" if hybrid else None,
                vector_queries=[VectorizableTextQuery(
                    text=probe["text"], k_nearest_neighbors=50, fields="content_vector",
                )] if hybrid else None,
            )
            hits = [
                {**{field: raw.get(field) for field in SELECT},
                 "score": raw.get("@search.score"),
                 "reranker_score": raw.get("@search.reranker_score"), "rank": rank}
                async for rank, raw in numbered(rows)
            ]
            result.update(hits=hits, total_count=await rows.get_count(), error=None)
    except Exception as error:
        result.update(hits=[], total_count=None, error=type(error).__name__)
    return {**result, "latency_ms": 1000 * (time.perf_counter() - started)}


async def numbered(rows: Any) -> Any:
    rank = 0
    async for row in rows:
        rank += 1
        yield rank, row


async def audit(args: argparse.Namespace) -> dict[str, Any]:
    source = args.comparisons.read_bytes()
    rows = [json.loads(line) for line in source.decode("utf-8-sig").splitlines() if line]
    selected = [row for row in rows if row["comparison_verdict"] in STATUSES]
    if args.only:
        unknown = set(args.only) - {row["id"] for row in selected}
        if unknown:
            raise ValueError(f"Unknown or non-target question IDs: {sorted(unknown)}")
        selected = [row for row in selected if row["id"] in args.only]
    if not selected or len({row["id"] for row in selected}) != len(selected):
        raise ValueError("Select nonempty, unique saved answer IDs")
    args.out_dir.mkdir(parents=True, exist_ok=True)
    manifest_path = args.out_dir / "manifest.json"
    if manifest_path.exists():
        raise ValueError("Refusing to overwrite an audit; use a fresh directory")
    manifest = {
        "started_at": datetime.now(UTC).isoformat(), "comparisons_sha256": hashlib.sha256(source).hexdigest(),
        "endpoint": args.endpoint, "index": args.index, "release": args.release,
        "top": args.top, "concurrency": args.concurrency,
        "selected_ids": [row["id"] for row in selected],
        "statuses": dict(Counter(row["comparison_verdict"] for row in selected)),
        "oracle_warning": "Reference-informed lookups are not fair retrieval or answer-quality scores",
        "scope": "Azure AI Search directly, no agent/KG scoping or answer generation",
        "source_passages_saved": True,
    }
    manifest_path.write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    gate = asyncio.Semaphore(args.concurrency)
    outcomes = []
    async with DefaultAzureCredential() as credential, SearchClient(
        args.endpoint, args.index, credential, retry_total=2,
        connection_timeout=10, read_timeout=30,
    ) as client:
        with (args.out_dir / "search-results.jsonl").open("x", encoding="utf-8") as stream:
            async def one(row: dict[str, Any]) -> None:
                for probe in probe_queries(row):
                    async with gate:
                        result = await query(client, row, probe, args.release, args.top)
                    stream.write(json.dumps(result, ensure_ascii=False) + "\n")
                    stream.flush()
                    outcomes.append(result)

            await asyncio.gather(*(one(row) for row in selected))
    summary = {
        "questions": len(selected), "queries": len(outcomes),
        "query_errors": dict(Counter(row["error"] for row in outcomes if row["error"])),
        "empty_queries": sum(not row["hits"] and not row["error"] for row in outcomes),
        "hits_saved": sum(len(row["hits"]) for row in outcomes),
        "unique_passages": len({hit["projection_id"] for row in outcomes for hit in row["hits"]}),
        "arms": dict(Counter(row["arm"] for row in outcomes)),
        "completed_at": datetime.now(UTC).isoformat(),
        "semantic_completeness": "not_scored",
    }
    (args.out_dir / "summary.json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
    return summary


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--comparisons", type=Path, required=True)
    parser.add_argument("--out-dir", type=Path, required=True)
    parser.add_argument("--endpoint", required=True)
    parser.add_argument("--index", required=True)
    parser.add_argument("--release", default="crv-v1")
    parser.add_argument("--only", action="append")
    parser.add_argument("--top", type=int, default=10)
    parser.add_argument("--concurrency", type=int, default=2)
    parser.add_argument("--confirm-live", action="store_true")
    args = parser.parse_args(argv)
    if not args.confirm_live or not 1 <= args.top <= 50 or not 1 <= args.concurrency <= 4:
        parser.error("Use --confirm-live, top 1-50 and concurrency 1-4")
    if not args.endpoint.startswith("https://"):
        parser.error("Search requires HTTPS")
    print(json.dumps(asyncio.run(audit(args))))
    return 0


if __name__ == "__main__":
    sys.exit(main())
